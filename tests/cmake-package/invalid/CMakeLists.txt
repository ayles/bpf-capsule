cmake_minimum_required(VERSION 3.24)
project(BpfCapsuleInvalidArgument C)

find_package(BpfCapsule CONFIG REQUIRED)
if(INVALID_CALL STREQUAL "wrong-llvm")
    set(BPF_CAPSULE_LLVM_AR "${CMAKE_COMMAND}")
    bpf_capsule_library(unused SOURCES ../smoke.c)
elseif(INVALID_CALL STREQUAL "library")
    bpf_capsule_library(unused TYPO SOURCES ../smoke.c)
elseif(INVALID_CALL STREQUAL "missing-library")
    bpf_capsule_object(unused OUTPUT unused.bpf.o SOURCES ../smoke.c LIBRARIES missing)
else()
    bpf_capsule_object(unused TYPO OUTPUT unused.bpf.o SOURCES ../smoke.c)
endif()
